  function moveFormal(index,dir){
    const target=index+dir;
    if(target<0||target>=state.model.formals.length)return;
    [state.model.formals[index],state.model.formals[target]]=[state.model.formals[target],state.model.formals[index]];
    state.model.items.forEach(item=>{
      [item.values[index],item.values[target]]=[item.values[target],item.values[index]];
      [item.valuePresent[index],item.valuePresent[target]]=[item.valuePresent[target],item.valuePresent[index]];
    });
    syncFormals();
    renderFormals();
    renderItems();
  }
