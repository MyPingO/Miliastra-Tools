      input.value=String(item.values[index]);
        });
        vals.append(row);
      });
      itemsBox.append(card);
    });
  };

  overlay.querySelector('[data-add-item]').addEventListener('click',()=>{
    addStatusItem();
    renderItems();
  });

  renderFormals();
  renderItems();

  overlay.querySelector('[data-close]').addEventListener('click',()=>{
    overlay.remove();overlay=null;state=null;
  });

  overlay.querySelector('[data-export]').addEventListener('click',()=>{
    try{
      const bytes=buildFile(state);
      downloadBytes(state.file.name.replace(/\.gia$/i,'')+' - edited.gia',bytes);
      overlay.querySelector('.sd-status').textContent=`Exported ${state.model.items.length} Status Item${state.model.items.length===1?'':'s'} and ${state.model.formals.length} Formal Variable${state.model.formals.length===1?'':'s'}.`;
    }catch(err){
      console.error(err);
      overlay.querySelector('.sd-status').textContent=`Export failed: ${err.message||err}`;
    }
  });
}

function escapeHtml(value){
  return String(value).replace(/[&<>"']/g,ch=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[ch]));
}

function startEditor(file,bytes,buildNew=false){
  const found=statusFile(bytes);
  if(!found)return false;
  state={file,bytes,model:found.m};
  if(buildNew){
    const m=state.model;
    const keep=m.items[0];
    const itemWrappers=new Set(m.items.map(i=>i.wrapperField));
    const keptContainer=m.container.filter(f=>!itemWrappers.has(f)||f===keep.wrapperField);
    m.container.splice(0,m.container.length,...keptContainer);
    if(m.containerField)m.containerField.v.raw=new Uint8Array();
    m.items=[keep];

    markName(keep,'New Status Item');
    setItemId(keep,1);
    setRefs(keep,[]);

    /* Build New starts semantically empty. The embedded template only supplies
       the component's required protobuf/UI skeleton; Formal Variables belong
       entirely to the user. */
    m.formals=[];
    if(Array.isArray(m.formalField?.v?.fields)){
      replaceRepeated(m.formalField.v.fields,501,2,[]);
      m.formalField.v.raw=new Uint8Array();
    }
    keep.values=[];
    keep.valuePresent=[];
    if(Array.isArray(keep.dataField?.v?.fields)){
      replaceRepeated(keep.dataField.v.fields,501,2,[]);
      keep.dataField.v.raw=new Uint8Array();
    }
  }
  render();
  return true;
}

function b64bytes(s){
  const raw=atob(s),out=new Uint8Array(raw.length);
  for(let i=0;i<raw.length;i++)out[i]=raw.charCodeAt(i);
  return out;
}
function replaceAllByteSequences(bytes,from,to){
  if(from.length!==to.length)throw Error('Temporary UI ID replacement changed varint length');
  const out=new Uint8Array(bytes);
  outer:for(let i=0;i<=out.length-from.length;i++){
    for(let j=0;j<from.length;j++)if(out[i+j]!==from[j])continue outer;
    out.set(to,i);
    i+=from.length-1;
  }
  return out;
}
function generateTemporaryUiIds(count,used){
  const taken=new Set(used),result=[];
  while(result.length<count){
    let random;
    if(globalThis.crypto?.getRandomValues){
      const b=new Uint32Array(1);crypto.getRandomValues(b);random=b[0]&0x000fffff;
    }else random=Math.floor(Math.random()*0x100000);
    const candidate=0x40000000|random;
    if(taken.has(candidate))continue;
    taken.add(candidate);result.push(candidate);
  }
  return result;
}
function createFreshStatusTemplateBytes(){
  let bytes=b64bytes(BUILD_TEMPLATE);
  const generated=generateTemporaryUiIds(BUILD_TEMPLATE_UI_IDS.length,BUILD_TEMPLATE_UI_IDS);
  BUILD_TEMPLATE_UI_IDS.forEach((id,i)=>{
    bytes=replaceAllByteSequences(bytes,vi(id),vi(generated[i]));
  });
  return bytes;
}

function installBuildCard(){
  if(document.getElementById('newStatusDisplayButton'))return;
  const sections=[...document.querySelectorAll('.build-section')];
  const gia=sections.find(section=>section.querySelector('h2')?.textContent.trim()==='UI Component Exports (.gia)');
  if(!gia)return;
  const grid=gia.querySelector('.build-grid')||gia;
  const card=document.createElement('div');
  card.className='build-card';
  card.tabIndex=0;
  card.setAttribute('role','button');
  card.innerHTML='<h3>Status Display Area</h3><p>Build status items, Unit Status links, and optional Formal Variables from scratch.</p><button id="newStatusDisplayButton" type="button">Status Display Area</button>';
  const button=card.querySelector('button');
  const open=()=>{
    try{
      const bytes=createFreshStatusTemplateBytes();
      const file=new File([bytes],'New Status Display Area.gia',{type:'application/octet-stream'});
      if(!startEditor(file,bytes,true))throw Error('Embedded Status Display Area template could not be parsed');
    }catch(err){
      console.error(err);
      if(typeof setStatus==='function')setStatus(`Could not create Status Display Area: ${err.message||err}`,'error');
      else alert(`Could not create Status Display Area:\n\n${err.message||err}`);
    }
  };
  button.addEventListener('click',event=>{event.preventDefault();event.stopPropagation();open()});
  card.addEventListener('click',event=>{
    if(event.target.closest('button,input,select,textarea,a'))return;
    open();
  });
  card.addEventListener('keydown',event=>{
    if(event.key==='Enter'||event.key===' '){event.preventDefault();open()}
  });
  grid.append(card);
}

const input=document.getElementById('fileInput');
if(input)input.addEventListener('change',async event=>{
  const file=input.files?.[0];
  if(!file)return;
  event.stopImmediatePropagation();
  try{
    const bytes=new Uint8Array(await file.arrayBuffer());
    if(startEditor(file,bytes,false)){
      input.value='';
      return;
    }
    await openFile(file);
  }catch(err){
    console.error(err);
    alert(`Could not open file:\n\n${err.message||err}`);
    input.value='';
  }
},true);

if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',installBuildCard);
else installBuildCard();

})();
