function escapeHtml(value){
  return String(value).replace(/[&<>"']/g,ch=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[ch]));
}

/* Status Display property 66 exists even when it contains zero Formal Variables.
   Prefer that exact property instead of requiring at least one definition. */
findFormalContainer=function(root){
  const exact=[];
  walk(root,fs=>{
    for(const field of fs){
      if(field.n!==66||field.w!==2||!Array.isArray(field.v?.fields))continue;
      const defs=all(field.v.fields,501,2).map(definitionInfo).filter(Boolean);
      exact.push({field,defs});
    }
  });
  if(exact.length)return exact.sort((a,b)=>b.defs.length-a.defs.length)[0];

  const candidates=[];
  walk(root,fs=>{
    if(!f0(fs,501,0)||!f0(fs,502,0)||!f0(fs,503,0)||!f0(fs,504,2))return;
    for(const field of fs){
      if(field.w!==2||!Array.isArray(field.v?.fields))continue;
      const defs=all(field.v.fields,501,2).map(definitionInfo).filter(Boolean);
      if(defs.length)candidates.push({field,defs});
    }
  });
  return candidates.length?candidates.sort((a,b)=>b.defs.length-a.defs.length)[0]:null;
};

function readMonitorReference(field){
  if(!Array.isArray(field?.v?.fields))return{scope:'none',code:null,name:''};
  const codeField=f0(field.v.fields,501,0),nameField=f0(field.v.fields,502,2);
  const code=codeField?.v;
  if(code==null||code===18446744073709551615n)return{scope:'none',code:null,name:''};
  const numeric=Number(code);
  return{
    scope:numeric===100?'player':numeric===101?'stage':'unknown',
    code:numeric,
    name:str(nameField)??''
  };
}

function setMonitorReference(scope,name){
  const field=state?.model?.monitorField;
  if(!field)return;
  if(scope==='none'){
    field.v.fields=[makeVarField(501,18446744073709551615n)];
  }else{
    const code=scope==='player'?100:scope==='stage'?101:null;
    if(code==null)throw Error('Unsupported Monitor Entity Variable source');
    field.v.fields=[
      makeVarField(501,code),
      {n:502,w:2,v:{raw:te.encode(String(name||'')),fields:null}}
    ];
  }
  field.v.raw=new Uint8Array();
  state.model.monitor=readMonitorReference(field);
}

/* Re-detect the model with the Monitor Entity Variable field attached.
   The Status Item container itself contains field 503 = monitor reference. */
findModel=function(payload){
  const root=parse(payload);
  if(!root)return null;
  let best=null;
  walk(root,fs=>{
    const items=all(fs,501,2).map(field=>{
      const item=itemInfo(field.v?.fields);
      if(item)item.wrapperField=field;
      return item;
    }).filter(Boolean);
    if(items.length<1)return;
    if(!best||items.length>best.items.length)best={root,container:fs,items};
  });
  if(!best)return null;
  walk(root,fs=>{
    for(const field of fs){
      if(field.w===2&&field.v?.fields===best.container)best.containerField=field;
    }
  });
  const fc=findFormalContainer(root);
  best.formalField=fc?.field||null;
  best.formals=(fc?.defs||[]).sort((a,b)=>a.id-b.id);
  best.monitorField=f0(best.container,503,2)||null;
  best.monitor=readMonitorReference(best.monitorField);
  best.items.forEach(item=>{
    const maps=readPairMaps(item.dataField,best.formals);
    item.values=maps.values;
    item.valuePresent=maps.present;
  });
  return best;
};

/* The component name is user-editable and Miliastra may suffix it on import.
   Detection therefore relies on the confirmed Status Item + monitor structure,
   not the literal string "Status Display Area". */
statusFile=function(bytes){
  if(!(bytes instanceof Uint8Array)||bytes.length<24)return null;
  const view=new DataView(bytes.buffer,bytes.byteOffset,20);
  if(view.getUint32(16,false)!==bytes.length-24)return null;
  const payload=bytes.slice(20,-4),m=findModel(payload);
  if(!m||!m.monitorField)return null;
  return{payload,m};
};

function b64bytes(s){
  const raw=atob(s),out=new Uint8Array(raw.length);
  for(let i=0;i<raw.length;i++)out[i]=raw.charCodeAt(i);
  return out;
}
function replaceAllByteSequences(bytes,from,to){
  if(from.length!==to.length)throw Error('Temporary UI ID replacement changed varint length');
  const out=new Uint8Array(bytes);
  outer:for(let i=0;i<=out.length-from.length;i++){
    for(let j=0;j<from.length;j++)if(out[i+j]!==from[j])continue outer;
    out.set(to,i);
    i+=from.length-1;
  }
  return out;
}
function generateTemporaryUiIds(count,used){
  const taken=new Set(used),result=[];
  while(result.length<count){
    let random;
    if(globalThis.crypto?.getRandomValues){
      const b=new Uint32Array(1);crypto.getRandomValues(b);random=b[0]&0x000fffff;
    }else random=Math.floor(Math.random()*0x100000);
    const candidate=0x40000000|random;
    if(taken.has(candidate))continue;
    taken.add(candidate);result.push(candidate);
  }
  return result;
}
