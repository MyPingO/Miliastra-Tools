        v=bytes.slice(p,p+4);p+=4;
      }else{
        let len;
        [len,p]=pv(bytes,p);
        len=Number(len);
        if(!Number.isSafeInteger(len)||len<0||p+len>bytes.length)return null;
        const raw=bytes.slice(p,p+len);p+=len;
        v={raw,fields:parse(raw)};
      }
      fields.push({n,w,v});
    }
    return p===bytes.length?fields:null;
  }catch{
    return null;
  }
}

function dirtyField(f){
  if(!f||f.w!==2)return false;
  if(f.v?.raw?.length===0)return true;
  return Array.isArray(f.v?.fields)&&f.v.fields.some(dirtyField);
}

function encField(f){
  const key=vi((f.n<<3)|f.w);
  if(f.w===0)return cat(key,vi(f.v));
  if(f.w===1||f.w===5)return cat(key,f.v);
  const body=f.v.fields&&dirtyField(f)?enc(f.v.fields):f.v.raw;
  return cat(key,vi(body.length),body);
}
const enc=fields=>cat(...fields.map(encField));

  cb(fs,path);
  fs.forEach((f,i)=>{

  });
}

function floatBytes(value){
  const b=new Uint8Array(4);
  new DataView(b.buffer).setFloat32(0,Number(value)||0,true);
  return b;
}

  return b instanceof Uint8Array&&b.length===4
    ?new DataView(b.buffer,b.byteOffset,b.byteLength).getFloat32(0,true)
    :0;
}

function valueWrapper(typeEnum,value){
  const t=typeInfo(typeEnum);
  let inner;
  if(t.kind==='int')inner=makeVarField(501,Math.max(0,Math.trunc(Number(value)||0)));
  else if(t.kind==='float')inner=makeRawField(501,5,floatBytes(value));
  else if(t.kind==='dynamic')inner=makeLenField(501,[{n:501,w:2,v:{raw:te.encode(String(value??'')),fields:null}}]);
  else inner={n:501,w:2,v:{raw:te.encode(String(value??'')),fields:null}};
  return [
    makeLenField(t.typed,[inner]),
    makeVarField(501,t.code),
    makeLenField(502,[makeVarField(501,18446744073709551615n)])
  ];
}

function readValue(wrapper,typeEnum){
  const t=typeInfo(typeEnum);
  try{
    const tf=f0(wrapper,t.typed,2),vf=f0(tf.v.fields,501);
    if(t.kind==='int')return Number(vf.v);
    if(t.kind==='float')return floatValue(vf.v);
    if(t.kind==='dynamic'){
      const nested=f0(vf.v.fields,501,2);
      return str(nested)??'';
    }
    return str(vf)??'';
  }catch{
    return defaultValue(typeEnum);
  }
}

function readPairMaps(dataField,formals){
  const byId=new Map();
  if(Array.isArray(dataField?.v?.fields)){
    for(const p of all(dataField.v.fields,501,2)){
      if(!Array.isArray(p.v?.fields))continue;
      const id=f0(p.v.fields,501,0),wrap=f0(p.v.fields,502,2);
      if(id&&Array.isArray(wrap?.v?.fields))byId.set(Number(id.v),wrap.v.fields);
    }
  }
  return {
    values:formals.map(v=>byId.has(v.id)?readValue(byId.get(v.id),v.typeEnum):defaultValue(v.typeEnum)),
    present:formals.map(v=>byId.has(v.id))
  };
}

function replaceRepeated(fs,n,w,replacements){
  if(!Array.isArray(fs))return;
  const first=fs.findIndex(f=>f.n===n&&f.w===w);
  const insertAt=first<0?0:first;
  let keptBefore=0;
  for(let i=0;i<insertAt;i++)if(!(fs[i].n===n&&fs[i].w===w))keptBefore++;
  const kept=fs.filter(f=>!(f.n===n&&f.w===w));
  kept.splice(keptBefore,0,...replacements);
  fs.splice(0,fs.length,...kept);
}

function findFormalContainer(root){
  const candidates=[];
  walk(root,fs=>{
    if(!f0(fs,501,0)||!f0(fs,502,0)||!f0(fs,503,0)||!f0(fs,504,2))return;
    for(const field of fs){
      if(field.w!==2||!Array.isArray(field.v?.fields))continue;
      const defs=all(field.v.fields,501,2).map(definitionInfo).filter(Boolean);
      if(defs.length)candidates.push({field,defs});
    }
  });
  if(!candidates.length)return null;
  return candidates.sort((a,b)=>b.defs.length-a.defs.length)[0];
}
