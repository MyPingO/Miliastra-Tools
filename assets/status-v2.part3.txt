ms=all(fs,501,2).map(field=>{
      const item=itemInfo(field.v?.fields);
      if(item)item.wrapperField=field;
      return item;
    }).filter(Boolean);
    if(items.length<1)return;
    if(!best||items.length>best.items.length)best={root,container:fs,items};
  });
  if(!best)return null;
  walk(root,fs=>{
    for(const field of fs){
      if(field.w===2&&field.v?.fields===best.container)best.containerField=field;
    }
  });
  const fc=findFormalContainer(root);
  best.formalField=fc?.field||null;
  best.formals=(fc?.defs||[]).sort((a,b)=>a.id-b.id);
  best.items.forEach(item=>{
    const maps=readPairMaps(item.dataField,best.formals);
    item.values=maps.values;
    item.valuePresent=maps.present;
  });
  return best;
}

function statusFile(bytes){
  if(!(bytes instanceof Uint8Array)||bytes.length<24)return null;
  const declared=new DataView(bytes.buffer,bytes.byteOffset,20).getUint32(16,false);
  if(declared!==bytes.length-24)return null;
  const payload=bytes.slice(20,-4);
  const m=findModel(payload);
  if(!m)return null;
  let has=false;
  walk(m.root,fs=>{
    for(const f of fs)if(f.w===2&&str(f)==='Status Display Area')has=true;
  });
  return has?{payload,m}:null;
}

function refField(id){return makeLenField(501,[makeVarField(501,Number(id))])}
function setRefs(item,refs){

  item.refsField.v.fields=refs.map(refField);
  item.refsField.v.raw=new Uint8Array();
}

function markName(item,name){
  item.name=String(name||'New Status Item');
  item.nameField.v.raw=te.encode(item.name);
  item.nameField.v.fields=null;
  item.nameWrap.v.raw=new Uint8Array();
}

function setItemId(item,id){
  item.id=Number(id);
  item.idField.v=item.id;
  item.wrapperField.v.raw=new Uint8Array();
}

let state=null,overlay=null;

function syncFormals(){
  if(!state)return;
  const m=state.model;
  m.formals.forEach((v,i)=>v.id=i+1);
  if(m.formalField){
    replaceRepeated(m.formalField.v.fields,501,2,m.formals.map((v,i)=>buildDefinition(v,i+1)));
    m.formalField.v.raw=new Uint8Array();
  }
  for(const item of m.items){
    while(item.values.length<m.formals.length)item.values.push(defaultValue(m.formals[item.values.length].typeEnum));
    while(item.valuePresent.length<m.formals.length)item.valuePresent.push(false);
    if(item.values.length>m.formals.length)item.values.length=m.formals.length;
    if(item.valuePresent.length>m.formals.length)item.valuePresent.length=m.formals.length;
    if(Array.isArray(item.dataField?.v?.fields)){
      const pairs=m.formals.flatMap((v,i)=>item.valuePresent[i]
        ?[buildPair(v,i+1,item.values[i]??defaultValue(v.typeEnum))]
        :[]);
      replaceRepeated(item.dataField.v.fields,501,2,pairs);
      item.dataField.v.raw=new Uint8Array();
    }
  }
}

function cloneFields(fs){
  const bytes=enc(fs);
  const clone=parse(bytes);
  if(!clone)throw Error('Could not clone Status Item template');
  return clone;
}

function blankItemFrom(source,id,name='New Status Item'){
  const fs=cloneFields(source.fs);
  const wrapper=makeLenField(501,fs);
  const item=itemInfo(fs);
  if(!item)throw Error('Status Item template could not be parsed');
  item.wrapperField=wrapper;
  markName(item,name);
  setItemId(item,id);
  setRefs(item,[]);
  if(Array.isArray(item.dataField?.v?.fields)){
    replaceRepeated(item.dataField.v.fields,501,2,[]);
    item.dataField.v.raw=new Uint8Array();
  }
  item.values=state.model.formals.map(v=>defaultValue(v.typeEnum));
  item.valuePresent=state.model.formals.map(()=>false);
  return item;
}

function addStatusItem(){
  const m=state.model;
  const source=m.items[0];
  if(!source)throw Error('No Status Item template is available');
  const id=Math.max(0,...m.items.map(i=>i.id))+1;
  const item=blankItemFrom(source,id,`Status Item ${id}`);
  let insertAt=m.container.length;
  for(let i=m.container.length-1;i>=0;i--){
    const f=m.container[i];
    if(f.n===501&&f.w===2&&itemInfo(f.v?.fields)){insertAt=i+1;break}
  }
  m.container.splice(insertAt,0,item.wrapperField);
  if(m.containerField)m.containerField.v.raw=new Uint8Array();
  m.items.push(item);
  return item;
}

function deleteStatusItem(item){
  const m=state.model;
  if(m.items.length<=1)return false;
  const wi=m.container.indexOf(item.wrapperField);
  if(wi>=0)m.container.splice(wi,1);
  if(m.containerField)m.containerField.v.raw=new Uint8Array();
  const ii=m.items.indexOf(item);
  if(ii>=0)m.items.splice(ii,1);
  return true;
}

function duplicateStatusItem(item){
  const m=state.model;
  const id=Math.max(0,...m.items.map(i=>i.id))+1;
  const fs=cloneFields(item.fs);
  const wrapper=makeLenField(501,fs);
  const copy=itemInfo(fs);
  if(!copy)throw Error('Could not duplicate Status Item');
  copy.wrapperField=wrapper;
  copy.values=[...item.values];
  copy.valuePresent=[...item.valuePresent];
  markName(copy,`${item.name} Copy`);
  setItemId(copy,id);
  let at=m.container.indexOf(item.wrapperField);
  if(at<0)at=m.container.length-1;
  m.container.splice(at+1,0,wrapper);
  if(m.containerField)m.containerField.v.raw=new Uint8Array();
  m.items.push(copy);
  return copy;
}

function buildFile(current){
  syncFormals();
  const payload=enc(current.model.root);
  const header=new Uint8Array(current.bytes.slice(0,20));
  const view=new DataView(header.buffer,header.byteOffset,header.byteLength);
  view.setUint32(0,20+payload.length,false);
  view.setUint32(16,payload.length,false);
  const out=cat(header,payload,current.bytes.slice(-4));
  if(out.length-4!==view.getUint32(0,false)||out.length-24!==view.getUint32(16,false))
    throw Error('Export size validation failed');
  if(!statusFile(out))
    throw Error('Export validation failed: generated GIA is not structurally valid');
  return out;
}

function parseId(v){
  const s=String(v).trim();
  if(!s)return null;
  if(!/^\d+$/.test(s))return null;
  const n=Number(s);
  return Number.isSafeInteger(n)&&n>=0?n:null;
}
